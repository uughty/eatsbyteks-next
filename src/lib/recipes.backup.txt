const fmt = (n: number) => (n >= 10 ? Math.round(n) : Math.round(n * 4) / 4).toString().replace(/\.25$/, "¼").replace(/\.5$/, "½").replace(/\.75$/, "¾").replace(/^0(?=[¼½¾])/, "");
export function scaledIngredients(recipe: Recipe, servings: number) {
  const list = amounts[recipe.id] ?? [];
  return recipe.ingredients.map((name, i) => {
    const a = list[i];
    return { name, amount: a ? `${fmt((a[0] * servings) / baseServings)}${a[1] ? " " + a[1] : ""}` : "" };
  });
}

export function recipeSwaps(recipe: Recipe) {
  return recipe.ingredients.flatMap((ing) => { const hit = swapTable.find(([k]) => ing.toLowerCase().includes(k)); return hit ? [{ ingredient: ing, swap: hit[1] }] : []; });
}
